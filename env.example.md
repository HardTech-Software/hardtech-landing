# Ejemplo de variables de entorno para HardTech Landing

# Copia este archivo como .env.local y configura tus valores

# Email principal de contacto

NEXT_PUBLIC_CONTACT_EMAIL=contacto@hardtech.cl

# Instagram

NEXT_PUBLIC_INSTAGRAM_HANDLE=@hardtech
NEXT_PUBLIC_INSTAGRAM_URL=https://www.instagram.com/hardtech

# URL del sitio (para metadata y sitemap)

NEXT_PUBLIC_SITE_URL=https://hardtech.cl

# Resend — proveedor de email para el formulario de contacto

# Obtén tu API key en https://resend.com/api-keys

RESEND_API_KEY=re_your_api_key_here

# Dirección que RECIBIRÁ los mensajes del formulario (bandeja del equipo)

CONTACT_TO_EMAIL=contacto@hardtech.cl

# Dirección que aparecerá como REMITENTE.

# Debe estar verificada en tu cuenta de Resend.

# Para pruebas puedes usar: onboarding@resend.dev (sandbox de Resend)

CONTACT_FROM_EMAIL=HardTech <noreply@hardtech.cl>
